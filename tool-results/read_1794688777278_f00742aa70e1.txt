   602→  );

   614→function ReviewForm({ productId, onClose }: { productId: string; onClose: () => void }) {
   615→  const [rating, setRating] = useState(5);
   616→  const [title, setTitle] = useState("");
   617→  const [body, setBody] = useState("");
   618→  const [name, setName] = useState("");
   619→  const [city, setCity] = useState("");
   620→  const [done, setDone] = useState(false);
   621→  const [loading, setLoading] = useState(false);
   622→
   623→  const submit = async (e: React.FormEvent) => {
   624→    e.preventDefault();
   625→    setLoading(true);
   626→    await fetch("/api/reviews", {
   627→      method: "POST",
   628→      headers: { "Content-Type": "application/json" },
   629→      body: JSON.stringify({ productId, authorName: name, authorCity: city, rating, title, body: body }),
   630→    });
   631→    setLoading(false);
   632→    setDone(true);
   633→    setTimeout(onClose, 2500);
   634→  };
   635→
   636→  if (done) {
   637→    return (
   638→      <div className="mt-4 rounded-md border border-gold/40 bg-gold/10 p-4 text-sm text-navy">
   639→        Thank you! Your review has been submitted and will appear once approved.
   640→      </div>
   641→    );
   642→  }
   643→
   644→  return (
   645→    <motion.form
   646→      initial={{ opacity: 0, height: 0 }}
   647→      animate={{ opacity: 1, height: "auto" }}
   648→      onSubmit={submit}
   649→      className="mt-4 space-y-3 overflow-hidden rounded-md border border-border bg-card p-4"
   650→    >
   651→      <div>
   652→        <label className="text-xs font-semibold uppercase tracking-wider text-navy">Your Rating</label>
   653→        <div className="mt-1.5 flex gap-1">
   654→          {[1, 2, 3, 4, 5].map((i) => (
   655→            <button key={i} type="button" onClick={() => setRating(i)} aria-label={`${i} stars`}>
   656→              <Star className={`h-6 w-6 ${i <= rating ? "fill-gold text-gold" : "fill-transparent text-muted-foreground/40"}`} />
   657→            </button>
   658→          ))}
   659→        </div>
   660→      </div>
   661→      <input required value={name} onChange={(e) => setName(e.target.value)} placeholder="Your name" className="h-10 w-full rounded-sm border border-border px-3 text-sm outline-none focus:border-gold" />
   662→      <input value={city} onChange={(e) => setCity(e.target.value)} placeholder="City (optional)" className="h-10 w-full rounded-sm border border-border px-3 text-sm outline-none focus:border-gold" />
   663→      <input required value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Review title" className="h-10 w-full rounded-sm border border-border px-3 text-sm outline-none focus:border-gold" />
   664→      <textarea required value={body} onChange={(e) => setBody(e.target.value)} placeholder="Share your experience..." rows={3} className="w-full rounded-sm border border-border p-3 text-sm outline-none focus:border-gold" />
   665→      <button type="submit" disabled={loading} className="rounded-sm bg-navy px-4 py-2 text-sm font-semibold text-ivory hover:bg-navy-light disabled:opacity-50">
   666→        {loading ? "Submitting..." : "Submit Review"}
   667→      </button>
   668→    </motion.form>
   669→  );
   670→}
